# 入口函数 main
# @param {dict} params - 输入项的集合
# @param {dict} context - 包含环境参数、应用ID等信息
# @return {dict} data - 返回的数据集合

import json
import re

def main(params: dict, context: dict) -> dict:
    try:
        input = params.get('param', {})
        intent = params.get('intent', {}) # 从节点输入项中提取名为 param 的参数
        data = {}
        # if '景点讲解' in input:

        data = flatten_dict(input)
        data["intent_params"] = json.dumps(
                {intent: input}, 
                ensure_ascii=False  # 支持中文
            )
        return data
    except Exception as e:
        return {
            "content" : str(e),
            "intent" : "其他",
            "paramsMap" : {}
        }


def flatten_dict(d, parent_key='', sep=''):
    """
    将嵌套字典展平为单层字典
    :param d: 嵌套字典
    :param parent_key: 父键，用于递归调用
    :param sep: 分隔符，用于连接键
    :return: 展平后的字典
    """
    items = []
    for k, v in d.items():
        new_key = k
        if isinstance(v, dict):
            items.extend(flatten_dict(v, new_key, sep=sep).items())
        else:
            items.append((new_key, v))
    return dict(items)

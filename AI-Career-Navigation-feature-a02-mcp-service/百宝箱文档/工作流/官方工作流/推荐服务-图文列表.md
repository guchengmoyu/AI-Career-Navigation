## 代码节点
### 代码节点
解析从知识库中召回的数据，并拼装出卡片需要的格式

```python
# 入口函数 main
# @param {dict} params - 输入项的集合
# @param {dict} context - 包含环境参数、应用ID等信息
# @return {dict} data - 返回的数据集合
import json
def main(params: dict, context: dict) -> dict:
    input = params.get('param', {}) # 从节点输入项中提取名为 param 的参数,是知识库召回的数据
    ##知识库召回的数据有冗余项，提取实际需要的内容，在content中
    cardlistdata = []
    for item in input:
        cardlistdata.append(item.get('content',{}))

    #准备拼接列表卡片的的数据
    list = []

    
    for carddatastr in cardlistdata:

        item = {}
        #抽取
        carddata = json.loads(carddatastr)
        #组装
        item['icon'] = carddata.get('icon','')
        item['title'] = carddata.get('title','')
        item['desc'] = carddata.get('desc','')
        item['actionType'] = carddata.get('actionType','')
        item['query'] = carddata.get('query','')
        item['jumpUrl'] = carddata.get('jumpUrl','')
        item['wxJumpUrl'] = carddata.get('wxJumpUrl','')

        #添加
        list.append(item)

    #构建返回数据，rowType是固定的，不用改
    data = {
        'rowType': 'COMMON_SERVICE',
        'list': list
    }
     #返回结果
    return data

```

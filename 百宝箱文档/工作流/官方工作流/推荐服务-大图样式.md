## 代码节点
### 代码节点
解析从知识库中召回的数据，并拼装出卡片需要的格式

```python
# 入口函数 main
# @param {dict} params - 输入项的集合
# @param {dict} context - 包含环境参数、应用ID等信息
# @return {dict} data - 返回的数据集合
import json
def main(params: dict, context: dict) -> dict:
    input = params.get('param', {}) # 从节点输入项中提取名为 param 的参数,是知识库召回的数据
    if input == []:
        return []
    #知识库召回的数据有冗余项，提取实际需要的内容，在content中
    cardDataList = []
    for item in input:
        cardDataList.append(item['content'])

    # 解析出卡片数据，知识库召回节点设置的只召回一个
    cardData = json.loads(cardDataList[0])
    imgUrl = cardData.get('image','')
    title = cardData.get('title','')

    #准备拼接按钮数据
    btnList = []

    #按钮的类型，只会是query、link、null之一，非null代表有此按钮
    jumpActionType = cardData.get('jumpActionType','null')
    detailsActionType = cardData.get('detailsActionType','null')
    #拼接查看详情按钮信息
    if detailsActionType != 'null':
        btnList.append({
            'btnText': '查看详情',
            'actionType': detailsActionType,
            'query': cardData.get('detailsQuery',''),
            'jumpUrl': cardData.get('detailsUrl',''),
            'wxJumpUrl': cardData.get('wxDetailsUrl',''),
            'type': 'primary'
        })

    #拼接立即前往按钮信息
    if jumpActionType != 'null':
        btnList.append({
            'btnText': '立即前往',
            'actionType': jumpActionType,
            'query': cardData.get('jumpQuery',''),
            'jumpUrl': cardData.get('jumpUrl',''),
            'wxJumpUrl': cardData.get('wxJumpUrl',''),
            'type': 'primary'
        })

    #根据按钮个数设置按钮样式
    if len(btnList) == 2:
        detailsbtn = btnList[0]
        detailsbtn['type'] = 'normal'


    # 构建返回数据，rowType是固定的，不用改
    data = {
        'imgUrl': imgUrl,
        'btnList': btnList,
        'title': title,
        'rowType': 'BANNER_CARD'
    }
    #返回结果
    return data

```
